# `orm: prisma` (Prisma 7, PostgreSQL)

Verified on ESM (Vitest) and CommonJS (Jest): unit tests + coverage ≥ 80 %, `test:e2e` against a real
PostgreSQL 18 container (Testcontainers, migrations applied with `prisma migrate deploy`), `build`, and the Docker
image answering `/health/ready` with the database `up`.

## Packages

- dependencies: `@prisma/client@7`, `@prisma/adapter-pg@7`, `pg@8`
- devDependencies: `prisma@7` (**pin the major** — `npm install prisma` resolves to a Prisma 8 release
  candidate), `dotenv@17` (the Prisma 7 CLI no longer loads `.env` itself), `@types/pg@8`, `testcontainers`,
  `@testcontainers/postgresql`

Keep `prisma` and `@prisma/client` on the same exact version (`npm view @prisma/client version` gives the stable
7.x; fallback `7.10.0`).

## Gotchas found verifying this file

- Don't run `prisma init`: it writes a `.env`, appends to `.gitignore`, and — observed with Prisma 7.10 — names its
  config `prisma7.config.ts`. Write the files below instead.
- `importFileExtension = "js"` in the generator is **required**: without it the generated client imports
  `./internal/class.ts`, which compiles but crashes at run time from `dist/` (`ERR_MODULE_NOT_FOUND`, observed in
  the container). With it, both ESM and CommonJS builds load.
- `moduleFormat = "esm"` for ESM, `"cjs"` for CommonJS.
- The generated client (`src/generated/prisma`) is not committed: `.gitignore` (via
  `iru-setup-typescript-gitignore`), `.prettierignore`, `.dockerignore`, `.oxlintrc.json` `ignorePatterns`, and the
  coverage exclusions all name it. CI runs `npx prisma generate` before lint; locally run it after every clone and
  schema change. `prisma generate` needs no `DATABASE_URL`.
- npm 11 blocks dependency install scripts by default (`npm install-scripts ls` lists `prisma`/`@prisma/engines`);
  Prisma 7 with a driver adapter needs no engine binary, so generate/migrate/run all worked with them blocked.
- The initial migration was generated with
  `npx prisma migrate diff --from-empty --to-schema prisma/schema.prisma --script` into
  `prisma/migrations/<timestamp>_init/migration.sql` (no database needed); later changes use
  `npx prisma migrate dev --name <change>` against a development database.
- Scripts to add: `"prisma:generate": "prisma generate"`, `"db:migrate": "prisma migrate deploy"`,
  `"db:migrate:dev": "prisma migrate dev"`.

### `src/database/prisma.service.ts`

```ts
import {
  Injectable,
  type OnModuleDestroy,
  type OnModuleInit,
} from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { PrismaPg } from '@prisma/adapter-pg';
import { PrismaClient } from '../generated/prisma/client.js';

/**
 * The Prisma client as a Nest provider. Prisma 7 requires a driver adapter; the connection string comes from the
 * validated `DATABASE_URL`. Connects at module init and disconnects on shutdown.
 */
@Injectable()
export class PrismaService
  extends PrismaClient
  implements OnModuleInit, OnModuleDestroy
{
  constructor(config: ConfigService) {
    super({
      adapter: new PrismaPg({
        connectionString: config.getOrThrow<string>('DATABASE_URL'),
      }),
    });
  }

  /** Opens the connection pool when the application starts. */
  async onModuleInit(): Promise<void> {
    await this.$connect();
  }

  /** Closes the connection pool when the application shuts down. */
  async onModuleDestroy(): Promise<void> {
    await this.$disconnect();
  }
}
```

### `src/books/books.repository.ts`

```ts
import { Injectable } from '@nestjs/common';
import { PrismaService } from '../database/prisma.service.js';
import type { Book, NewBook } from './book.entity.js';

/** Storage for books, backed by PostgreSQL through Prisma. */
@Injectable()
export class BooksRepository {
  constructor(private readonly prisma: PrismaService) {}

  /**
   * Lists books ordered by id.
   *
   * @param limit - Maximum number of books to return.
   * @param offset - Number of books to skip.
   * @returns The requested page, possibly empty.
   */
  findAll(limit: number, offset: number): Promise<readonly Book[]> {
    return this.prisma.book.findMany({
      orderBy: { id: 'asc' },
      take: limit,
      skip: offset,
    });
  }

  /**
   * Finds one book.
   *
   * @param id - Identifier of the book.
   * @returns The book, or `undefined` when no book has that id.
   */
  async findById(id: number): Promise<Book | undefined> {
    return (await this.prisma.book.findUnique({ where: { id } })) ?? undefined;
  }

  /**
   * Stores a new book.
   *
   * @param data - The book's fields.
   * @returns The stored book with its assigned id.
   */
  create(data: NewBook): Promise<Book> {
    return this.prisma.book.create({
      data: { title: data.title, author: data.author },
    });
  }

  /**
   * Deletes one book.
   *
   * @param id - Identifier of the book.
   * @returns `true` when a book was deleted, `false` when none had that id.
   */
  async delete(id: number): Promise<boolean> {
    const { count } = await this.prisma.book.deleteMany({ where: { id } });
    return count > 0;
  }
}
```

### `src/books/books.repository.spec.ts`

```ts
import { Test } from '@nestjs/testing';
import { PrismaService } from '../database/prisma.service.js';
import { BooksRepository } from './books.repository.js';

describe('BooksRepository (Prisma)', () => {
  const dune = { id: 1, title: 'Dune', author: 'Frank Herbert' };
  const prisma = {
    book: {
      findMany: vi.fn(),
      findUnique: vi.fn(),
      create: vi.fn(),
      deleteMany: vi.fn(),
    },
  };
  let repository: BooksRepository;

  beforeEach(async () => {
    vi.clearAllMocks();
    const moduleRef = await Test.createTestingModule({
      providers: [
        BooksRepository,
        { provide: PrismaService, useValue: prisma },
      ],
    }).compile();
    repository = moduleRef.get(BooksRepository);
  });

  it('pages through books ordered by id', async () => {
    prisma.book.findMany.mockResolvedValue([dune]);

    await expect(repository.findAll(10, 20)).resolves.toEqual([dune]);
    expect(prisma.book.findMany).toHaveBeenCalledWith({
      orderBy: { id: 'asc' },
      take: 10,
      skip: 20,
    });
  });

  it('finds a book by id', async () => {
    prisma.book.findUnique.mockResolvedValue(dune);

    await expect(repository.findById(1)).resolves.toEqual(dune);
  });

  it('returns undefined for an unknown id', async () => {
    prisma.book.findUnique.mockResolvedValue(null);

    await expect(repository.findById(9)).resolves.toBeUndefined();
  });

  it('creates a book from explicit fields', async () => {
    prisma.book.create.mockResolvedValue(dune);

    await expect(
      repository.create({ title: 'Dune', author: 'Frank Herbert' }),
    ).resolves.toEqual(dune);
    expect(prisma.book.create).toHaveBeenCalledWith({
      data: { title: 'Dune', author: 'Frank Herbert' },
    });
  });

  it('reports whether a book was deleted', async () => {
    prisma.book.deleteMany
      .mockResolvedValueOnce({ count: 1 })
      .mockResolvedValueOnce({ count: 0 });

    await expect(repository.delete(1)).resolves.toBe(true);
    await expect(repository.delete(1)).resolves.toBe(false);
  });
});
```

### `src/health/health.controller.ts`

```ts
import { Controller, Get } from '@nestjs/common';
import {
  HealthCheck,
  HealthCheckService,
  PrismaHealthIndicator,
  type HealthCheckResult,
} from '@nestjs/terminus';
import { PrismaService } from '../database/prisma.service.js';

/**
 * Liveness and readiness probes under `/health`, for the container `HEALTHCHECK`, Kubernetes probes, and load
 * balancers. Liveness never checks dependencies (a database outage must not restart every replica); readiness
 * checks what the instance needs to serve traffic.
 */
@Controller('health')
export class HealthController {
  constructor(
    private readonly health: HealthCheckService,
    private readonly prismaHealth: PrismaHealthIndicator,
    private readonly prisma: PrismaService,
  ) {}

  /**
   * `GET /health/live` - the process is up and its event loop answers.
   *
   * @returns The Terminus result (`200` with `status: "ok"`).
   */
  @Get('live')
  @HealthCheck()
  live(): Promise<HealthCheckResult> {
    return this.health.check([]);
  }

  /**
   * `GET /health/ready` - the instance may receive traffic.
   *
   * @returns The Terminus result; `503` when the database does not answer within 1.5 s.
   */
  @Get('ready')
  @HealthCheck()
  ready(): Promise<HealthCheckResult> {
    return this.health.check([
      () =>
        this.prismaHealth.pingCheck('database', this.prisma).withTimeout(1500),
    ]);
  }
}
```

### `src/health/health.controller.spec.ts`

```ts
import {
  HealthCheckService,
  PrismaHealthIndicator,
  type HealthCheckResult,
} from '@nestjs/terminus';
import { Test } from '@nestjs/testing';
import { PrismaService } from '../database/prisma.service.js';
import { HealthController } from './health.controller.js';

describe('HealthController', () => {
  const ok: HealthCheckResult = {
    status: 'ok',
    info: {},
    error: {},
    details: {},
  };
  const health = { check: vi.fn() };
  const attempt = { withTimeout: vi.fn() };
  const prismaHealth = { pingCheck: vi.fn() };
  const prisma = {};
  let controller: HealthController;

  beforeEach(async () => {
    vi.clearAllMocks();
    health.check.mockResolvedValue(ok);
    const moduleRef = await Test.createTestingModule({
      controllers: [HealthController],
      providers: [
        { provide: HealthCheckService, useValue: health },
        { provide: PrismaHealthIndicator, useValue: prismaHealth },
        { provide: PrismaService, useValue: prisma },
      ],
    }).compile();
    controller = moduleRef.get(HealthController);
  });

  it('reports liveness without checking dependencies', async () => {
    await expect(controller.live()).resolves.toEqual(ok);
    expect(health.check).toHaveBeenCalledWith([]);
  });

  it('reports readiness by pinging the database', async () => {
    health.check.mockImplementation(
      async (checks: readonly (() => unknown)[]) => {
        await Promise.all(checks.map((check) => check()));
        return ok;
      },
    );
    prismaHealth.pingCheck.mockReturnValue(attempt);
    attempt.withTimeout.mockResolvedValue({ database: { status: 'up' } });

    await expect(controller.ready()).resolves.toEqual(ok);
    expect(prismaHealth.pingCheck).toHaveBeenCalledWith('database', prisma);
    expect(attempt.withTimeout).toHaveBeenCalledWith(1500);
  });
});
```

### `src/config/env.schema.ts`

```ts
import { z } from 'zod';

/**
 * Environment variables the application reads, validated once at bootstrap by `ConfigModule`.
 *
 * Every value arrives as a string, so numbers are coerced. A variable without a default is required: the
 * application refuses to start with a `Config validation error` naming it instead of failing on first use.
 */
export const envSchema = z.object({
  NODE_ENV: z
    .enum(['development', 'production', 'test'])
    .default('development'),
  PORT: z.coerce.number().int().min(0).max(65_535).default(3000),
  DATABASE_URL: z.string().min(1),
});

/** The validated, typed environment produced by {@link envSchema}. */
export type Env = z.output<typeof envSchema>;
```

### `src/config/env.schema.spec.ts`

```ts
import { envSchema } from './env.schema.js';

describe('envSchema', () => {
  const DATABASE_URL = 'postgresql://app:secret@localhost:5432/app';

  it('applies the defaults when only the required variables are set', () => {
    expect(envSchema.parse({ DATABASE_URL })).toEqual({
      NODE_ENV: 'development',
      PORT: 3000,
      DATABASE_URL,
    });
  });

  it('coerces PORT from its string form', () => {
    expect(envSchema.parse({ DATABASE_URL, PORT: '8080' }).PORT).toBe(8080);

  it('rejects a PORT that is not a number', () => {
    expect(envSchema.safeParse({ DATABASE_URL, PORT: 'abc' }).success).toBe(
      false,
    );
  });

  it('requires DATABASE_URL', () => {
    expect(envSchema.safeParse({}).success).toBe(false);
  });
});
```

### `src/app.module.ts`

```ts
import { Module, ValidationPipe } from '@nestjs/common';
import { ConfigModule } from '@nestjs/config';
import { APP_PIPE } from '@nestjs/core';
import { BooksModule } from './books/books.module.js';
import { envSchema } from './config/env.schema.js';
import { DatabaseModule } from './database/database.module.js';
import { HealthModule } from './health/health.module.js';

/**
 * Root module: validated configuration, the global validation pipe, health checks, and the feature modules.
 *
 * The `ValidationPipe` is registered here as an `APP_PIPE` provider rather than with `app.useGlobalPipes()` in
 * `main.ts`, so every test that imports `AppModule` validates requests exactly like production does.
 */
@Module({
  imports: [
    ConfigModule.forRoot({ isGlobal: true, validationSchema: envSchema }),
    DatabaseModule,
    HealthModule,
    BooksModule,
  ],
  providers: [
    {
      provide: APP_PIPE,
      useFactory: (): ValidationPipe =>
        new ValidationPipe({
          whitelist: true,
          forbidNonWhitelisted: true,
          transform: true,
        }),
    },
  ],
})
export class AppModule {}
```

### `test/global-setup.ts`

```ts
import { execFileSync } from 'node:child_process';
import { PostgreSqlContainer } from '@testcontainers/postgresql';

/**
 * Starts one throwaway PostgreSQL container for the whole e2e run, applies the Prisma migrations to it, and
 * exports its URL as `DATABASE_URL` before any test file imports `AppModule` (test workers start after this and
 * inherit the environment). Needs a running Docker daemon.
 *
 * @returns The teardown function that stops the container.
 */
export default async function setup(): Promise<() => Promise<void>> {
  const postgres = await new PostgreSqlContainer('postgres:18-alpine').start();
  process.env['DATABASE_URL'] = postgres.getConnectionUri();
  execFileSync('npx', ['prisma', 'migrate', 'deploy'], {
    stdio: 'inherit',
    env: process.env,
  });

  return async () => {
    await postgres.stop();
  };
}
```

## CommonJS (Jest) e2e setup

`test/jest-e2e.json` gains `"globalSetup": "<rootDir>/global-setup.ts"` and `"globalTeardown": "<rootDir>/global-teardown.ts"` (and keeps the `.js` `moduleNameMapper` from `app-templates.md` — the generated client imports `./x.js`). Jest runs the two hooks in separate module scopes, so the container handle travels on `globalThis`:

### `test/global-setup.ts`

```ts
import { execFileSync } from 'node:child_process';
import {
  PostgreSqlContainer,
  type StartedPostgreSqlContainer,
} from '@testcontainers/postgresql';

declare global {
  var postgresContainer: StartedPostgreSqlContainer | undefined;
}

/**
 * Starts one throwaway PostgreSQL container for the whole e2e run, applies the Prisma migrations to it, and
 * exports its URL as `DATABASE_URL` before any test file imports `AppModule`. `global-teardown.ts` stops it.
 */
export default async function setup(): Promise<void> {
  const postgres = await new PostgreSqlContainer('postgres:18-alpine').start();
  globalThis.postgresContainer = postgres;
  process.env['DATABASE_URL'] = postgres.getConnectionUri();
  execFileSync('npx', ['prisma', 'migrate', 'deploy'], {
    stdio: 'inherit',
    env: process.env,
  });
}
```

### `test/global-teardown.ts`

```ts
/** Stops the PostgreSQL container `global-setup.ts` started. */
export default async function teardown(): Promise<void> {
  await globalThis.postgresContainer?.stop();
}
```

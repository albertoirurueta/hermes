# Persistence templates: Alembic, the integration tier, and the other shapes

Read the section matching the run's shape:

| Shape | Sections |
|---|---|
| `database: postgresql`, `orm: sqlmodel` (default) | Alembic + Integration tier |
| `orm: sqlalchemy` | Alembic + Integration tier, then **replace** the files listed under "orm: sqlalchemy" |
| `database: sqlite` | Alembic (no integration tier), plus "database: sqlite" |
| `database: none` | only "database: none" (no Alembic, no `app/db.py`, no `compose.yaml`) |

Migrations (irurueta docs: `backend/fastapi/sql-databases-with-sqlmodel.adoc`,
`backend/flask/database-migrations.adoc`, `backend/pyramid/sqlalchemy-transactions-and-migrations.adoc`):

- `alembic init -t async migrations` is the starting point; the files below **replace** what it generates (delete
  its `migrations/README`). `migrations/env.py` uses the async engine with `NullPool`, `compare_type=True`,
  `render_as_batch` on SQLite, and reads the URL from `sqlalchemy.url` (tests set it) or `DatabaseSettings`
  (`APP_DATABASE_URL` only — a migration job needs no JWT secret or other application setting).
- Naming convention set before the first table (`pk_book`, `uq_book_isbn`), so constraint names are identical on
  every engine.
- `script.py.mako` is modern-typed (`collections.abc.Sequence`, `X | None`) and imports `sqlmodel.sql.sqltypes`
  (the autogenerated `AutoString` needs it; a bare `import sqlmodel` fails `mypy --strict` with `Module has no
  attribute "sql"`). The `alembic.ini` post-write hooks run `ruff check --fix` + `ruff format` on every new revision,
  which also drops the unused `sqlmodel` import from revisions that don't need it — so generated revisions pass
  `ruff check .` (incl. `UP`) and `mypy .` without a per-file ignore beyond `D`.
- Revisions use explicit ids: `uv run alembic revision --autogenerate --rev-id 0002 -m "<what>"` (files sort as
  `0002_<slug>.py`). Autogenerate is a draft: review every file (renames, NOT NULL on populated tables, server
  defaults); never edit a revision that has been applied anywhere.
- `0001_create_book_table.py` is written verbatim (it is what autogenerate produced against PostgreSQL 18 and is
  dialect-neutral; set `Create Date` to today). Verified: `alembic upgrade head` → `alembic check` ("No new upgrade
  operations detected.") → one head → `downgrade base` → `upgrade head` on PostgreSQL 18 and SQLite.
- Migrations run as a one-shot step (`compose.yaml` `migrate` service, a Kubernetes Job), never at app start-up.

### `migrations/env.py`

```python
"""Alembic environment: migrates the database the application itself is configured for.

The URL comes from ``sqlalchemy.url`` when a caller sets it (tests do), otherwise
from ``APP_DATABASE_URL`` (``DatabaseSettings``: only the connection, so migration
jobs need no other application secret).
"""

import asyncio
from logging.config import fileConfig

from alembic import context
from sqlalchemy import pool
from sqlalchemy.engine import Connection
from sqlalchemy.ext.asyncio import create_async_engine
from sqlmodel import SQLModel

import app.models  # noqa: F401  (registers every table on SQLModel.metadata)
from app.settings import DatabaseSettings

config = context.config
if config.config_file_name is not None:
    fileConfig(config.config_file_name, disable_existing_loggers=False)

target_metadata = SQLModel.metadata


def _database_url() -> str:
    url = config.get_main_option("sqlalchemy.url")
    return url or DatabaseSettings().database_url.get_secret_value()


def run_migrations_offline() -> None:
    """Emit the migration SQL to stdout (``alembic upgrade head --sql``)."""
    url = _database_url()
    context.configure(
        url=url,
        target_metadata=target_metadata,
        compare_type=True,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
        render_as_batch=url.startswith("sqlite"),
    )
    with context.begin_transaction():
        context.run_migrations()


def _run_migrations(connection: Connection) -> None:
    context.configure(
        connection=connection,
        target_metadata=target_metadata,
        compare_type=True,
        render_as_batch=connection.dialect.name == "sqlite",
    )
    with context.begin_transaction():
        context.run_migrations()


async def _run_async_migrations() -> None:
    engine = create_async_engine(_database_url(), poolclass=pool.NullPool)
    async with engine.connect() as connection:
        await connection.run_sync(_run_migrations)
    await engine.dispose()


def run_migrations_online() -> None:
    """Connect to the database and apply the migrations."""
    asyncio.run(_run_async_migrations())


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()
```

```python
"""Declarative base shared by every table mapping."""

from sqlalchemy import MetaData
from sqlalchemy.orm import DeclarativeBase


class Base(DeclarativeBase):
    """Base class of the table mappings.

    The naming convention gives every constraint a deterministic name (set before the
    first migration): without it PostgreSQL and SQLite name constraints differently and
    Alembic cannot drop them reliably later.
    """

    metadata = MetaData(
        naming_convention={
            "ix": "ix_%(column_0_label)s",
            "uq": "uq_%(table_name)s_%(column_0_name)s",
            "ck": "ck_%(table_name)s_%(constraint_name)s",
            "fk": "fk_%(table_name)s_%(column_0_name)s_%(referred_table_name)s",
            "pk": "pk_%(table_name)s",
        }
    )
```

### `app/models/book.py (orm: sqlalchemy)`

```python
"""The ``book`` table mapping and the create/public/update API schemas."""

from typing import Annotated

from pydantic import BaseModel, ConfigDict, Field, field_validator
from sqlalchemy import String
from sqlalchemy.orm import Mapped, mapped_column

from app.models.base import Base

Title = Annotated[str, Field(min_length=1, max_length=200)]
Author = Annotated[str, Field(min_length=1, max_length=200)]
Isbn = Annotated[
    str,
    Field(
        min_length=13,
        max_length=13,
        pattern=r"^\d{13}$",
        description="ISBN-13, digits only.",
    ),
]


class Book(Base):
    """The ``book`` table. Endpoints never return it: they answer `BookPublic`."""

    __tablename__ = "book"

    id: Mapped[int] = mapped_column(primary_key=True)
    title: Mapped[str] = mapped_column(String(200))
    author: Mapped[str] = mapped_column(String(200))
    isbn: Mapped[str] = mapped_column(String(13), unique=True)


class BookCreate(BaseModel):
    """Request body for creating a book. Unknown fields are rejected."""

    model_config = ConfigDict(extra="forbid")

    title: Title
    author: Author
    isbn: Isbn


class BookPublic(BaseModel):
    """A book as returned by the API, read from a :class:`Book` row."""

    model_config = ConfigDict(from_attributes=True)

    id: int
    title: str
    author: str
    isbn: str


class BookUpdate(BaseModel):
    """Partial update: only the fields sent are changed, ``null`` is rejected."""

    model_config = ConfigDict(extra="forbid")

    title: Title | None = None
    author: Author | None = None

    @field_validator("title", "author", mode="after")
    @classmethod
    def reject_null(cls, value: str | None) -> str:
        """Allow omitting a field, but not sending ``null`` (the columns are NOT NULL).

        Args:
            value: The validated value.

        Returns:
            The value unchanged.

        Raises:
            ValueError: If the client sent an explicit ``null``.
        """
        if value is None:
            msg = "may be omitted but not null"
            raise ValueError(msg)
        return value
```

### `app/db.py (orm: sqlalchemy)`

```python
"""Database plumbing: one async engine per process and a session factory per engine."""

from sqlalchemy import event
from sqlalchemy.engine.interfaces import DBAPIConnection
from sqlalchemy.ext.asyncio import (
    AsyncEngine,
    AsyncSession,
    async_sessionmaker,
    create_async_engine,
)
from sqlalchemy.pool import ConnectionPoolEntry, StaticPool


def create_engine(url: str, *, echo: bool = False) -> AsyncEngine:
    """Create the application's async engine.

    PostgreSQL gets ``pool_pre_ping`` so connections dropped by the server or a
    proxy are replaced transparently. SQLite (tests, local runs) gets foreign keys
    switched on to behave like PostgreSQL, and an in-memory database is shared
    through a single connection so every session sees the same tables.

    Args:
        url: SQLAlchemy async URL (``postgresql+asyncpg://...`` or ``sqlite+aiosqlite://...``).
        echo: Whether to log every SQL statement.

    Returns:
        A new engine; dispose of it with ``await engine.dispose()`` on shutdown.
    """
    if not url.startswith("sqlite"):
        return create_async_engine(url, echo=echo, pool_pre_ping=True)
    if url.endswith((":memory:", "://")):
        engine = create_async_engine(url, echo=echo, poolclass=StaticPool)
    else:
        engine = create_async_engine(url, echo=echo)
    event.listen(engine.sync_engine, "connect", _enable_sqlite_foreign_keys)
    return engine


def create_sessionmaker(engine: AsyncEngine) -> async_sessionmaker[AsyncSession]:
    """Create the session factory bound to ``engine``.

    ``expire_on_commit=False`` keeps loaded attributes readable after the commit, which
    async code needs because a lazy reload would require an implicit await.

    Args:
        engine: The engine sessions connect through.

    Returns:
        A factory producing :class:`AsyncSession` objects.
    """
    return async_sessionmaker(engine, expire_on_commit=False)


def _enable_sqlite_foreign_keys(
    dbapi_connection: DBAPIConnection, _connection_record: ConnectionPoolEntry
) -> None:
    cursor = dbapi_connection.cursor()
    cursor.execute("PRAGMA foreign_keys=ON")
    cursor.close()
```

### `app/data/books.py (orm: sqlalchemy)`

```python
"""SQL implementation of :class:`app.services.ports.BookRepository`."""

from collections.abc import Sequence

from sqlalchemy import select
from sqlalchemy.exc import IntegrityError
from sqlalchemy.ext.asyncio import AsyncSession

from app.models import Book
from app.services.errors import DuplicateIsbnError


class SqlBookRepository:
    """Stores books through an :class:`AsyncSession`; flushes, never commits."""

    def __init__(self, session: AsyncSession) -> None:
        """Create the repository.

        Args:
            session: The request's session (it owns the transaction).
        """
        self._session = session

    async def get(self, book_id: int) -> Book | None:
        """Return the book with this id, or ``None``."""
        return await self._session.get(Book, book_id)

    async def get_by_isbn(self, isbn: str) -> Book | None:
        """Return the book with this ISBN, or ``None``."""
        result = await self._session.scalars(select(Book).where(Book.isbn == isbn))
        return result.first()

    async def list_page(self, *, offset: int, limit: int) -> Sequence[Book]:
        """Return up to ``limit`` books ordered by id, skipping the first ``offset``."""
        statement = select(Book).order_by(Book.id).offset(offset).limit(limit)
        result = await self._session.scalars(statement)
        return result.all()

    async def save(self, book: Book) -> Book:
        """Insert or update ``book`` and return it with its generated id.

        Raises:
            DuplicateIsbnError: If the unique ISBN constraint rejects the row (a
                concurrent request inserted the same ISBN after the service checked).
        """
        self._session.add(book)
        try:
            await self._session.flush()
        except IntegrityError as exc:
            raise DuplicateIsbnError(book.isbn) from exc
        await self._session.refresh(book)
        return book

    async def delete(self, book: Book) -> None:
        """Delete ``book``."""
        await self._session.delete(book)
        await self._session.flush()
```

### `app/services/books.py (orm: sqlalchemy)`

```python
"""Book use cases."""

from collections.abc import Sequence

from app.models import Book, BookCreate, BookUpdate
from app.services.errors import BookNotFoundError, DuplicateIsbnError
from app.services.ports import BookRepository


class BookService:
    """Create, read, update and delete books, enforcing the catalogue's rules."""

    def __init__(self, repository: BookRepository) -> None:
        """Create the service.

        Args:
            repository: Where books are stored.
        """
        self._repository = repository

    async def create(self, data: BookCreate) -> Book:
        """Add a book to the catalogue.

        Args:
            data: The validated request body.

        Returns:
            The stored book, with its id.

        Raises:
            DuplicateIsbnError: If another book already has this ISBN.
        """
        if await self._repository.get_by_isbn(data.isbn) is not None:
            raise DuplicateIsbnError(data.isbn)
        return await self._repository.save(Book(**data.model_dump()))

    async def get(self, book_id: int) -> Book:
        """Return one book.

        Args:
            book_id: The book's id.

        Returns:
            The book.

        Raises:
            BookNotFoundError: If no book has this id.
        """
        book = await self._repository.get(book_id)
        if book is None:
            raise BookNotFoundError(book_id)
        return book

    async def list_page(self, *, offset: int, limit: int) -> Sequence[Book]:
        """Return one page of books ordered by id.

        Args:
            offset: How many books to skip.
            limit: The maximum number of books to return.

        Returns:
            The books on the page (possibly none).
        """
        return await self._repository.list_page(offset=offset, limit=limit)

    async def update(self, book_id: int, changes: BookUpdate) -> Book:
        """Change the fields present in ``changes``, leaving the others untouched.

        Args:
            book_id: The book's id.
            changes: The fields to change.

        Returns:
            The updated book.

        Raises:
            BookNotFoundError: If no book has this id.
        """
        book = await self.get(book_id)
        for field, value in changes.model_dump(exclude_unset=True).items():
            setattr(book, field, value)
        return await self._repository.save(book)

    async def delete(self, book_id: int) -> None:
        """Remove a book from the catalogue.

        Args:
            book_id: The book's id.

        Raises:
            BookNotFoundError: If no book has this id.
        """
        await self._repository.delete(await self.get(book_id))
```

```python
from typing import Annotated

from pydantic import BaseModel, ConfigDict, Field, field_validator

Title = Annotated[str, Field(min_length=1, max_length=200)]
Author = Annotated[str, Field(min_length=1, max_length=200)]
Isbn = Annotated[
    str,
    Field(
        min_length=13,
        max_length=13,
        pattern=r"^\d{13}$",
        description="ISBN-13, digits only.",
    ),
]


class Book(BaseModel):
    """A stored book; ``id`` is assigned by the repository."""

    id: int | None = None
    title: Title
    author: Author
    isbn: Isbn


class BookCreate(BaseModel):
    """Request body for creating a book. Unknown fields are rejected."""

    model_config = ConfigDict(extra="forbid")

    title: Title
    author: Author
    isbn: Isbn


class BookPublic(BaseModel):
    """A book as returned by the API."""

    model_config = ConfigDict(from_attributes=True)

    id: int
    title: str
    author: str
    isbn: str


class BookUpdate(BaseModel):
    """Partial update: only the fields sent are changed, ``null`` is rejected."""

    model_config = ConfigDict(extra="forbid")

    title: Title | None = None
    author: Author | None = None

    @field_validator("title", "author", mode="after")
    @classmethod
    def reject_null(cls, value: str | None) -> str:
        """Allow omitting a field, but not sending ``null``.

        Args:
            value: The validated value.

        Returns:
            The value unchanged.

        Raises:
            ValueError: If the client sent an explicit ``null``.
        """
        if value is None:
            msg = "may be omitted but not null"
            raise ValueError(msg)
        return value
```

### `app/services/books.py (database: none)`

```python
"""Book use cases."""

from collections.abc import Sequence

from app.models import Book, BookCreate, BookUpdate
from app.services.errors import BookNotFoundError, DuplicateIsbnError
from app.services.ports import BookRepository


class BookService:
    """Create, read, update and delete books, enforcing the catalogue's rules."""

    def __init__(self, repository: BookRepository) -> None:
        """Create the service.

        Args:
            repository: Where books are stored.
        """
        self._repository = repository

    async def create(self, data: BookCreate) -> Book:
        """Add a book to the catalogue.

        Args:
            data: The validated request body.

        Returns:
            The stored book, with its id.

        Raises:
            DuplicateIsbnError: If another book already has this ISBN.
        """
        if await self._repository.get_by_isbn(data.isbn) is not None:
            raise DuplicateIsbnError(data.isbn)
        return await self._repository.save(Book(**data.model_dump()))

    async def get(self, book_id: int) -> Book:
        """Return one book.

        Args:
            book_id: The book's id.

        Returns:
            The book.

        Raises:
            BookNotFoundError: If no book has this id.
        """
        book = await self._repository.get(book_id)
        if book is None:
            raise BookNotFoundError(book_id)
        return book

    async def list_page(self, *, offset: int, limit: int) -> Sequence[Book]:
        """Return one page of books ordered by id.

        Args:
            offset: How many books to skip.
            limit: The maximum number of books to return.

        Returns:
            The books on the page (possibly none).
        """
        return await self._repository.list_page(offset=offset, limit=limit)

    async def update(self, book_id: int, changes: BookUpdate) -> Book:
        """Change the fields present in ``changes``, leaving the others untouched.

        Args:
            book_id: The book's id.
            changes: The fields to change.

        Returns:
            The updated book.

        Raises:
            BookNotFoundError: If no book has this id.
        """
        book = await self.get(book_id)
        changed = book.model_copy(update=changes.model_dump(exclude_unset=True))
        return await self._repository.save(changed)

    async def delete(self, book_id: int) -> None:
        """Remove a book from the catalogue.

        Args:
            book_id: The book's id.

        Raises:
            BookNotFoundError: If no book has this id.
        """
        await self._repository.delete(await self.get(book_id))
```

### `app/data/books.py (database: none)`

```python
"""In-memory implementation of :class:`app.services.ports.BookRepository`.

State lives in one worker process and is lost on restart: replace this adapter with a
database-backed one before running several workers or replicas.
"""

from collections.abc import Sequence

from app.models import Book
from app.services.errors import DuplicateIsbnError


class InMemoryBookRepository:
    """Stores books in a dict keyed by id."""

    def __init__(self, capacity: int) -> None:
        """Create an empty repository.

        Args:
            capacity: The maximum number of books kept.
        """
        self._capacity = capacity
        self._books: dict[int, Book] = {}
        self._next_id = 1

    async def get(self, book_id: int) -> Book | None:
        """Return the book with this id, or ``None``."""
        return self._books.get(book_id)

    async def get_by_isbn(self, isbn: str) -> Book | None:
        """Return the book with this ISBN, or ``None``."""
        return next((b for b in self._books.values() if b.isbn == isbn), None)

    async def list_page(self, *, offset: int, limit: int) -> Sequence[Book]:
        """Return up to ``limit`` books ordered by id, skipping the first ``offset``."""
        return [self._books[key] for key in sorted(self._books)][
            offset : offset + limit
        ]

    async def save(self, book: Book) -> Book:
        """Insert or update ``book`` and return it with its id.

        Raises:
            DuplicateIsbnError: If another stored book has the same ISBN.
            OverflowError: If the catalogue is full.
        """
        other = await self.get_by_isbn(book.isbn)
        if other is not None and other.id != book.id:
            raise DuplicateIsbnError(book.isbn)
        if book.id is None:
            if len(self._books) >= self._capacity:
                msg = f"the catalogue is full ({self._capacity} books)"
                raise OverflowError(msg)
            book = book.model_copy(update={"id": self._next_id})
            self._next_id += 1
        assert book.id is not None  # noqa: S101 (narrowing for the type checker)
        self._books[book.id] = book
        return book

    async def delete(self, book: Book) -> None:
        """Delete ``book``."""
        if book.id is not None:
            self._books.pop(book.id, None)
```

```python
"""Application factory: routers, exception handlers and the lifespan, nothing else."""

from collections.abc import AsyncIterator
from contextlib import asynccontextmanager
from typing import TypedDict

from fastapi import FastAPI, Request, status
from fastapi.responses import JSONResponse

from app import __version__
from app.data.books import InMemoryBookRepository
from app.routers import books, health
from app.services.errors import BookNotFoundError, DuplicateIsbnError
from app.settings import Settings, get_settings


class State(TypedDict):
    """Process-wide resources created by :func:`lifespan` (see ``request.state``)."""

    book_repository: InMemoryBookRepository


@asynccontextmanager
async def lifespan(app: FastAPI) -> AsyncIterator[State]:
    """Create the process-wide resources on start-up.

    Runs once per worker process. Dependency overrides do not apply inside a
    lifespan, so the settings provider is looked up in ``app.dependency_overrides``
    explicitly (tests override it).

    Args:
        app: The application being started.

    Yields:
        The lifespan state.
    """
    settings: Settings = app.dependency_overrides.get(get_settings, get_settings)()
    yield {"book_repository": InMemoryBookRepository(capacity=settings.max_books)}


def create_app() -> FastAPI:
    """Build the application.

    Returns:
        A new :class:`FastAPI` instance with its routers and exception handlers.
    """
    app = FastAPI(title="<title>", version=__version__, lifespan=lifespan)
    app.include_router(health.router)
    app.include_router(books.router, prefix="/api/v1")

    @app.exception_handler(BookNotFoundError)
    async def book_not_found(_request: Request, exc: BookNotFoundError) -> JSONResponse:
        return JSONResponse(
            status_code=status.HTTP_404_NOT_FOUND, content={"detail": str(exc)}
        )

    @app.exception_handler(DuplicateIsbnError)
    async def duplicate_isbn(
        _request: Request, exc: DuplicateIsbnError
    ) -> JSONResponse:
        return JSONResponse(
            status_code=status.HTTP_409_CONFLICT, content={"detail": str(exc)}
        )

    return app


app = create_app()
```

# xtask template — man pages, completions, CLI reference

`cargo xtask docs` (alias in `.cargo/config.toml`) generates everything from the same clap `Command` the binary
uses, as clap_mangen's documentation recommends (a dev task, not a flag on the shipped program).

Conditional markers (`// >>> <condition>` … `// <<< <condition>` in Rust, `# >>> …` in TOML) and the `__Token__`
placeholders work as described in SKILL.md's "Reference files" section; the marker lines themselves are never
written. A heading suffixed `(condition)` is written only when the condition holds. Every file below was rendered
and verified on 2026-10-16 for `ui: none` and `ui: tui` (Rust 1.99.1; MSRV toolchains 1.85.0 for `ui: none`, 1.88.0 for `ui: tui`) — see SKILL.md
"Verification notes".

### `xtask/src/main.rs`

```rust
//! `cargo xtask` — development tasks that need the command tree but don't belong in the shipped binary.
//!
//! `cargo xtask docs` writes, from the same clap `Command` the binary uses:
//! - man pages to `target/dist/man/` (one per command, via `clap_mangen`),
//! - completion scripts to `target/dist/completions/` (bash, zsh, fish, PowerShell, elvish),
//! - the CLI reference page to `docs/modules/ROOT/pages/cli-reference.adoc` when the Antora module
//!   exists, otherwise to `target/dist/cli-reference.adoc`.
//!
//! `cargo xtask docs --check` fails when the committed CLI reference is out of date (run it in CI).

use std::fmt::Write as _;
use std::io::{self, Write as _};
use std::path::{Path, PathBuf};
use std::process::ExitCode;

use __crate__::BIN_NAME;
use __crate__::cli::Cli;
use __crate__::commands::schema;
use clap::{CommandFactory, Parser, Subcommand};
use clap_complete::Shell;
use serde_json::Value;

/// Development tasks for this repository.
#[derive(Debug, Parser)]
#[command(name = "xtask")]
struct Args {
    #[command(subcommand)]
    task: Task,
}

#[derive(Debug, Subcommand)]
enum Task {
    /// Generate man pages, shell completions and the CLI reference from the command tree.
    Docs {
        /// Only verify that the committed CLI reference is up to date; write nothing.
        #[arg(long)]
        check: bool,
    },
}

fn main() -> ExitCode {
    let Args {
        task: Task::Docs { check },
    } = Args::parse();
    match docs(&workspace_root(), check) {
        Ok(()) => ExitCode::SUCCESS,
        Err(error) => {
            let _ = writeln!(io::stderr(), "xtask: error: {error}");
            ExitCode::FAILURE
        }
    }
}

fn workspace_root() -> PathBuf {
    Path::new(env!("CARGO_MANIFEST_DIR"))
        .parent()
        .map_or_else(|| PathBuf::from("."), Path::to_path_buf)
}

fn docs(root: &Path, check: bool) -> io::Result<()> {
    let antora_pages = root.join("docs/modules/ROOT/pages");
    let dist = root.join("target/dist");
    let antora = root.join("docs/modules/ROOT").is_dir();
    let reference_path = if antora {
        antora_pages.join("cli-reference.adoc")
    } else {
        dist.join("cli-reference.adoc")
    };
    let reference = cli_reference();
    if check && !antora {
        return writeln!(
            io::stdout(),
            "no docs/modules/ROOT: no committed CLI reference to check"
        );
    }
    if check {
        let current = std::fs::read_to_string(&reference_path).unwrap_or_default();
        if current != reference {
            return Err(io::Error::other(format!(
                "{} is out of date; run `cargo xtask docs` and commit the result",
                reference_path.display()
            )));
        }
        return writeln!(io::stdout(), "{} is up to date", reference_path.display());
    }

    let mut out = io::stdout().lock();
    let man_dir = dist.join("man");
    std::fs::create_dir_all(&man_dir)?;
    clap_mangen::generate_to(Cli::command(), &man_dir)?;
    writeln!(out, "man pages   -> {}", man_dir.display())?;

    let completions_dir = dist.join("completions");
    std::fs::create_dir_all(&completions_dir)?;
    for shell in [
        Shell::Bash,
        Shell::Zsh,
        Shell::Fish,
        Shell::PowerShell,
        Shell::Elvish,
    ] {
        clap_complete::generate_to(shell, &mut Cli::command(), BIN_NAME, &completions_dir)?;
    }
    writeln!(out, "completions -> {}", completions_dir.display())?;

    if let Some(parent) = reference_path.parent() {
        std::fs::create_dir_all(parent)?;
    }
    std::fs::write(&reference_path, reference)?;
    writeln!(out, "reference   -> {}", reference_path.display())
}

/// Renders the CLI reference page from `schema`'s command tree (plus each command's help examples).
fn cli_reference() -> String {
    let cli = Cli::command();
    let tree = schema::tree(&cli);
    let mut page = String::new();
    let _ = writeln!(page, "= CLI reference");
    let _ = writeln!(
        page,
        "// Generated by `cargo xtask docs` from src/cli.rs — do not edit by hand.\n"
    );
    let _ = writeln!(page, "`{BIN_NAME}` — {}\n", text(&tree["summary"]));
    let _ = writeln!(
        page,
        "Run `{BIN_NAME} schema` for this information as JSON.\n"
    );

    let _ = writeln!(page, "== Global flags\n");
    flag_table(&mut page, &tree["globalFlags"]);

    let _ = writeln!(page, "== Commands\n");
    for command in tree["commands"].as_array().into_iter().flatten() {
        let name = text(&command["name"]);
        let _ = writeln!(page, "=== `{BIN_NAME} {name}`\n");
        let _ = writeln!(page, "{}\n", text(&command["summary"]));
        if let Some(args) = command["args"].as_array().filter(|a| !a.is_empty()) {
            let _ = writeln!(
                page,
                ".Arguments\n[cols=\"1,1,1,3\"]\n|===\n|Argument |Type |Required |Description\n"
            );
            for arg in args {
                let _ = writeln!(
                    page,
                    "|`{}` |{} |{} |{}",
                    text(&arg["name"]).to_uppercase(),
                    kind(arg),
                    if arg["required"] == true { "yes" } else { "no" },
                    cell(&arg["description"])
                );
            }
            let _ = writeln!(page, "|===\n");
        }
        if command["flags"].as_array().is_some_and(|f| !f.is_empty()) {
            let _ = writeln!(page, ".Flags");
            flag_table(&mut page, &command["flags"]);
        }
        let safety = &command["safety"];
        let traits: Vec<&str> = [
            ("readOnly", "read-only"),
            ("destructive", "destructive"),
            ("idempotent", "idempotent"),
        ]
        .iter()
        .filter(|(key, _)| safety[*key] == true)
        .map(|(_, label)| *label)
        .collect();
        let codes: Vec<String> = command["exitCodes"]
            .as_array()
            .into_iter()
            .flatten()
            .map(ToString::to_string)
            .collect();
        let _ = writeln!(page, "Safety:: {}", traits.join(", "));
        let _ = writeln!(page, "Exit codes:: {}\n", codes.join(", "));
        if let Some(examples) = cli
            .find_subcommand(&name)
            .and_then(clap::Command::get_after_help)
        {
            let _ = writeln!(
                page,
                "[source,shell]\n----\n{}\n----\n",
                examples.to_string().trim()
            );
        }
    }

    let _ = writeln!(
        page,
        "== Exit codes\n\n[cols=\"1,2,5\"]\n|===\n|Code |Name |Meaning\n"
    );
    for exit in tree["exitCodes"].as_array().into_iter().flatten() {
        let _ = writeln!(
            page,
            "|{} |`{}` |{}",
            exit["code"],
            text(&exit["name"]),
            cell(&exit["meaning"])
        );
    }
    let _ = writeln!(page, "|===");
    page
}

fn flag_table(page: &mut String, flags: &Value) {
    let _ = writeln!(
        page,
        "[cols=\"2,2,2,4\"]\n|===\n|Flag |Type |Environment |Description\n"
    );
    for flag in flags.as_array().into_iter().flatten() {
        let name = match flag["short"].as_str() {
            Some(short) => format!("`{short}`, `{}`", text(&flag["name"])),
            None => format!("`{}`", text(&flag["name"])),
        };
        let env = flag["env"]
            .as_str()
            .map_or_else(String::new, |e| format!("`{e}`"));
        let _ = writeln!(
            page,
            "|{name} |{} |{env} |{}",
            kind(flag),
            cell(&flag["description"])
        );
    }
    let _ = writeln!(page, "|===\n");
}

fn kind(arg: &Value) -> String {
    let mut kind = text(&arg["type"]);
    if let Some(values) = arg["values"].as_array() {
        let values: Vec<String> = values.iter().map(|v| format!("`{}`", text(v))).collect();
        kind = format!("{kind}: {}", values.join(", "));
    }
    if let Some(default) = arg["default"].as_str() {
        kind = format!("{kind} (default `{default}`)");
    }
    kind
}

fn text(value: &Value) -> String {
    value.as_str().unwrap_or_default().to_owned()
}

fn cell(value: &Value) -> String {
    text(value).replace('|', "\\|")
}
```
